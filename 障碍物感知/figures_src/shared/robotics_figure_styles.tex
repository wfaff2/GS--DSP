% Shared TikZ styles for robotics/control paper figures.

\tikzset{
    rfPanel/.style={
        draw=black!50,
        line width=0.55pt,
        rounded corners=2pt,
        fill=white,
        inner sep=4pt
    },
    rfBluePanel/.style={rfPanel, fill=RFBlueLight},
    rfOrangePanel/.style={rfPanel, fill=RFOrangeLight},
    rfCorePanel/.style={rfPanel, fill=RFBeige},
    rfGreenPanel/.style={rfPanel, fill=RFGreenLight},
    rfGrayPanel/.style={rfPanel, fill=RFGray},
    rfBlock/.style={
        draw=black!55,
        line width=0.45pt,
        rounded corners=1.5pt,
        fill=white,
        align=center,
        inner sep=2.4pt,
        font=\footnotesize,
        minimum height=7.5mm
    },
    rfSmallBlock/.style={
        rfBlock,
        font=\scriptsize,
        minimum height=6.2mm,
        inner sep=1.8pt
    },
    rfBlueBlock/.style={rfBlock, draw=RFBlue!90, fill=RFBlueLight},
    rfOrangeBlock/.style={rfBlock, draw=RFOrange!95, fill=RFOrangeLight},
    rfGreenBlock/.style={rfBlock, draw=RFGreen!95, fill=RFGreenLight},
    rfAmberBlock/.style={rfBlock, draw=RFAmberDark, fill=RFAmberLight, line width=0.65pt},
    rfVioletBlock/.style={rfBlock, draw=RFVioletDark, fill=RFVioletLight, line width=0.65pt},
    rfGrayBlock/.style={rfBlock, draw=black!45, fill=RFGray},
    rfArrow/.style={
        -{Latex[length=2mm,width=1.55mm]},
        line width=0.55pt,
        draw=black!75
    },
    rfThickArrow/.style={
        -{Latex[length=2.3mm,width=1.9mm]},
        line width=0.75pt,
        draw=black!80
    },
    rfDashedArrow/.style={
        -{Latex[length=1.8mm,width=1.4mm]},
        line width=0.5pt,
        dashed,
        draw=black!55
    },
    rfFeedback/.style={
        -{Latex[length=2mm,width=1.55mm]},
        line width=0.55pt,
        draw=black!60,
        dashed
    },
    rfPanelTitle/.style={font=\bfseries\footnotesize, align=left, text=RFDarkText},
    rfPanelLetter/.style={font=\bfseries\large, align=center, text=RFDarkText},
    rfTinyLabel/.style={font=\scriptsize, align=center, text=RFDarkText},
    rfMathNote/.style={font=\scriptsize, align=center, text=black!80}
}
